function(moe_runtime_variant suffix)
    add_library(moe-memory-runtime${suffix} STATIC ${MOE_MEMORY_SOURCES})
    target_compile_features(moe-memory-runtime${suffix} PUBLIC cxx_std_17)
    target_include_directories(moe-memory-runtime${suffix} PUBLIC "${MOE_RUNTIME_ROOT}/include")
    target_link_libraries(moe-memory-runtime${suffix} PUBLIC ${CMAKE_DL_LIBS})
    add_library(moe-expert-slot-runtime${suffix} STATIC ${MOE_EXPERT_SOURCES})
    target_compile_features(moe-expert-slot-runtime${suffix} PUBLIC cxx_std_17)
    target_include_directories(moe-expert-slot-runtime${suffix} PUBLIC
        "${MOE_RUNTIME_ROOT}/include" "${MOE_RUNTIME_ROOT}/../../ggml/include")
    target_link_libraries(moe-expert-slot-runtime${suffix} PUBLIC moe-memory-runtime${suffix})
    set_property(TARGET moe-memory-runtime${suffix} PROPERTY INTERFACE_MOE_RUNTIME_VARIANT "normal${suffix}")
    set_property(TARGET moe-memory-runtime${suffix} APPEND PROPERTY COMPATIBLE_INTERFACE_STRING MOE_RUNTIME_VARIANT)
    if(suffix)
        target_compile_definitions(moe-memory-runtime${suffix} PUBLIC MOE_RUNTIME_PROFILE=1)
        target_compile_definitions(moe-expert-slot-runtime${suffix} PUBLIC MOE_RUNTIME_PROFILE=1)
    endif()
    foreach(target moe-memory-runtime${suffix} moe-expert-slot-runtime${suffix})
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endforeach()
endfunction()
moe_runtime_variant("")
moe_runtime_variant("-profile")

option(MOE_BUILD_RAW_ADAPTERS "Build raw OpenCL/FastRPC adapters without device profiles" OFF)

        target_link_libraries(moe-memory-opencl${suffix} PUBLIC moe-memory-runtime${suffix} ${OpenCL_LIBRARIES})

            target_compile_definitions(moe-raw-adapter${suffix} PRIVATE CL_TARGET_OPENCL_VERSION=300)
            target_include_directories(moe-raw-adapter${suffix} PUBLIC ${OpenCL_INCLUDE_DIRS}
                PRIVATE ${HEXAGON_SDK_ROOT}/incs ${HEXAGON_SDK_ROOT}/incs/stddef)
            target_link_libraries(moe-raw-adapter${suffix} PUBLIC moe-memory-opencl${suffix})
        endif()
    endforeach()
endif()
